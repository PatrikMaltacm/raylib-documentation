Começando com node-raylib

npm install raylib
const r = require('raylib');

r.InitWindow(800, 450, "Olá Mundo");
r.SetTargetFPS(60);

while (!r.WindowShouldClose()) {
    r.BeginDrawing();
    r.ClearBackground(r.RAYWHITE);
    r.DrawText("node-raylib funciona!", 190, 200, 20, r.LIGHTGRAY);
    r.EndDrawing();
}

r.CloseWindow();
